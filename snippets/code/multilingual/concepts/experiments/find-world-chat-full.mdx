```python Python
import os
import re

from dotenv import load_dotenv
from galileo import GalileoLogger
from galileo.experiments import run_experiment
from galileo.openai import openai

# Load environment variables (Galileo project, API keys, etc)
load_dotenv(override=True)

# Experiment data
dataset = [{ "input": "Varn" }, { "input": "Drath" }]
worlds_db = [
    {
        "name": "Varn",
        "description": "Varn is a tidally locked moon orbiting a gas giant; its habitable zone is a narrow band of eternal dusk where life has adapted to constant twilight.",
    },
    {
        "name": "Drath",
        "description": "Drath is a high-gravity planet covered in dense jungle, where evolution favors thick carapaces and low, crawling vegetation.",
    },
]

# Create a logger instance 
logger = GalileoLogger(log_stream="Fictional Worlds Logstream")

# Create a log session for the experiment
logger.start_session(name="Fictional Worlds Experiment")

def find_fictional_world(name: str) -> str:
    """Returns a short description of a fictional world matching `name`"""

    for w in worlds_db:
        if re.search(name, w["name"], flags=re.IGNORECASE):
            return w["description"]
    return "No matches found"

# The experiment's custom function
def llm_call(world_name: str):
    """Our experiment function for invoking the llm"""

    # Pre-emptively call the tool on behalf of the LLM
    world_desc = find_fictional_world(world_name)

    # set up the user request with information for the LLM.
    prompt = (
        f"Humorously describe {world_name} in a single sentence. "
        f"Here's the world description: {world_desc}"
    )

    # Create a new trace context in our logging session, 
    logger.add_trace(name="LLM call", input=prompt)
    # and add a span for the tool call
    logger.add_tool_span(
      name="Search worlds", 
      input=world_name, 
      output=world_desc
    )

    # invoke the LLM 
    request = openai.chat.completions.create(
        model="gpt-4o",
        messages=[{"role": "user", "content": prompt}],
    )

    # Print the response to your terminal
    response = request.choices[0].message.content
    print(response)

    # Conclude and flush the current trace
    logger.conclude(output=response)
    logger.flush()

    return response

# run the experiment
results = run_experiment(
    project=os.environ.get("GALILEO_PROJECT"),
    experiment_name="Fictional Worlds Experiment",
    dataset=dataset,
    function=llm_call,
)
```

```typescript TypeScript
import { configDotenv } from "dotenv";
import { GalileoLogger, runExperiment, wrapOpenAI } from "galileo";
import OpenAI from "openai";

// Load environment variables
configDotenv();

// Create a wrapped instance of OpenAI for typescript
const openai = wrapOpenAI(new OpenAI({ apiKey: process.env.OPENAI_API_KEY }));

// Experiment data
const dataset = [{ input: "Varn" }, { input: "Drath" }];
const worldsDb = [
    {
        name: "Varn",
        description: "Varn is a tidally locked moon orbiting a gas giant; its habitable zone is a narrow band of eternal dusk where life has adapted to constant twilight.",
    },
    {
        name: "Drath",
        description: "Drath is a high-gravity planet covered in dense jungle, where evolution favors thick carapaces and low, crawling vegetation.",
    },
];

// Create a logger instance 
const logger = new GalileoLogger(log_stream="Fictional Worlds Logstream");

// Create a log session for the experiment
logger.startSession({ name: "Fictional Worlds Experiment" });

/** Returns a short description of a fictional world matching `name` */ 
const findFictionalWorld = (name: string): string => {
  const match = worldsDb.find((w) => new RegExp(name, "i").test(w.name));
  return match?.description ?? "No matches found";
}

/** Our experiment function for invoking the llm */
async function llmCall({ input: worldName }: Record<string, string>) {

  // Pre-emptively call the tool on behalf of the LLM
  const worldDesc = findFictionalWorld(worldName);

  // set up the initial user request
  const prompt = `Humorously describe ${worldName} in a single sentence. 
  Here's the world description: ${worldDesc}`;

  // Create a new trace context in our logging session, 
  logger.startTrace({ name: "LLM call", input: prompt });
  // and add a span for the tool call
  logger.addToolSpan({ 
    name: "Search worlds", 
    input: worldName, 
    output: worldDesc
  });

  // invoke the LLM 
  const request = await openai.chat.completions.create({
    model: "gpt-4o",
    messages: [{ role: "user", content: prompt }],
  });
  
  // Print the response to your terminal
  const response = request.choices[0].message.content
  console.log(response);

  // Conclude and flush the current trace
  logger.conclude({ output: response });
  logger.flush();

  return response;
}

// run the experiment (note: `runExperiment` is an async function)
runExperiment({ 
    project: process.env.GALILEO_PROJECT,
    name: "Fictional Worlds Experiment"
    dataset: dataset,
    function: llmCall
})
```
