```python Python
from galileo.experiments import run_experiment
from galileo import GalileoMetrics

# Dataset with generated_output — no prompt template needed
dataset = [
    {
        "input": "What is the capital of France?",
        "generated_output": "The capital of France is Paris.",
        "ground_truth": "Paris",  # Optional — enables Ground Truth Adherence metric
    },
    {
        "input": "Explain quantum computing in one sentence.",
        "generated_output": "Quantum computing uses qubits to perform calculations exponentially faster than classical computers.",
        # No ground_truth here — metrics like completeness and context_adherence still work without it
    },
]

results = run_experiment(
    "evaluate-my-output",
    dataset=dataset,
    metrics=[
        GalileoMetrics.completeness,        # Works without ground_truth
        GalileoMetrics.context_adherence,    # Works without ground_truth
    ],
    project="my-project",
)
```
