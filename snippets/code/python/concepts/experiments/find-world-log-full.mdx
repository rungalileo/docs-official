```python Python
import os
import re
from dotenv import load_dotenv

# Galileo dependencies
from galileo import log
from galileo.experiments import run_experiment

# Load .env variables
load_dotenv(override=True)

# Experiment data
dataset = [{"input": "Varn"}, {"input": "Drath"}]
worlds_db = [
    {
        "name": "Varn",
        "description": (
            "Varn is a tidally locked moon orbiting a gas giant; its habitable "
            "zone is a narrow band of eternal dusk where life has adapted to constant twilight."
        ),
    },
    {
        "name": "Drath",
        "description": (
            "Drath is a high-gravity planet covered in dense jungle, "
            "where evolution favors thick carapaces and low, crawling vegetation."
        ),
    },
]

# Create a tool function
@log(span_type="tool")
def find_fictional_world(name: str) -> str:
    """Returns a short description of a fictional world matching `name`"""

    for w in worlds_db:
        if re.search(name, w["name"], flags=re.IGNORECASE):
            return w["description"]
    return "No matches found"

# Simulate an LLM call
@log(span_type="llm")
def describe_world(world_name: str):
    desc = find_fictional_world(world_name)
    return desc

# Define the custom function for the experiment
@log(span_type="workflow")
def llm_call(world_name: str) -> str:
    """Simulate an LLM call"""

    response = describe_world(world_name=world_name)
    print(f"LLM Response: {response}")

    return response


# run the experiment
results = run_experiment(
    project=os.environ.get("GALILEO_PROJECT"),
    experiment_name="Contrived FW Agent XP",
    dataset=dataset,
    function=llm_call,
)
```
