```python Python
import os
from opentelemetry.sdk.trace import TracerProvider
from galileo.otel import GalileoSpanProcessor, add_galileo_span_processor
from galileo_a2a import A2AInstrumentor

# Set up Galileo-backed OTel tracing
provider = TracerProvider()
processor = GalileoSpanProcessor(
    project=os.environ.get("GALILEO_PROJECT", "a2a-distributed-tracing"),
    logstream=os.environ.get("GALILEO_LOG_STREAM", "dev"),
)
add_galileo_span_processor(provider, processor)

# Instrument A2A with agent name for identification
a2a_instrumentor = A2AInstrumentor()
a2a_instrumentor.instrument(tracer_provider=provider, agent_name="orchestrator")

# Optional: combine with other instrumentors for full-stack tracing
from opentelemetry.instrumentation.langchain import LangchainInstrumentor

LangchainInstrumentor().instrument(tracer_provider=provider)
```
