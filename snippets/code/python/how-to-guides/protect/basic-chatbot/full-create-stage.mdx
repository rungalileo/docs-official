```python create_central_stage.py
import os
from dotenv import load_dotenv

from galileo.stages import create_protect_stage
from galileo_core.schemas.protect.rule import Rule, RuleOperator
from galileo_core.schemas.protect.ruleset import Ruleset
from galileo_core.schemas.protect.action import OverrideAction
from galileo_core.schemas.protect.stage import StageType

# Load environment and create a fresh project
load_dotenv()
project_name = os.environ.get("GALILEO_PROJECT")
if not project_name:
    raise RuntimeError("Please set GALILEO_PROJECT in your environment or .env file.")

# Define the toxicity guardrail: rule + override action
toxicity_rule = Rule(metric="input_toxicity", operator=RuleOperator.gt, target_value=0.01)
action = OverrideAction(
    type="OVERRIDE",
    choices=[
        "This input was blocked due to toxicity.",
        "Content flagged: please rephrase your input.",
        "Input rejected: it violated the safety policy.",
    ],
)

# Create a ruleset from the toxicity rule and action
ruleset = Ruleset(rules=[toxicity_rule], action=action)

# Create a central Protect Stage with the ruleset
stage = create_protect_stage(
    project_name=project_name,
    name="Toxicity Stage",
    stage_type=StageType.central,
    prioritized_rulesets=[ruleset],
)

print(f"✅ Created stage: {stage.name} (id={stage.id}) in project '{project_name}'.")
```
