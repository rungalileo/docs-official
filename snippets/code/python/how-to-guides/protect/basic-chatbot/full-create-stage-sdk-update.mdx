"""create_central_stage.py – complete script for creating a project, an empty Protect Stage,
and then attaching a toxicity guardrail using SDK Ruleset objects
"""

import os
from dotenv import load_dotenv

from galileo.projects import get_project, create_project
from galileo.stages import create_protect_stage, update_protect_stage
from galileo_core.schemas.protect.rule import Rule, RuleOperator
from galileo_core.schemas.protect.ruleset import Ruleset
from galileo_core.schemas.protect.action import OverrideAction
from galileo_core.schemas.protect.stage import StageType

# Load environment and create a fresh project
load_dotenv()
project_name = os.environ.get("GALILEO_PROJECT")
if not project_name:
    raise RuntimeError("Please set GALILEO_PROJECT in your environment or .env file.")

# Ensure the project name is new for this tutorial
if get_project(name=project_name):
    raise RuntimeError(
        f"Project '{project_name}' already exists.\n"
        "➡️ For this tutorial, edit your .env and set GALILEO_PROJECT to a new, unused name.\n"
        "This script will create a new project with that name automatically."
    )

# Create the project
project = create_project(name=project_name)
print(f"✅ Created project: {project.name} (id={project.id})")

# Define the toxicity guardrail: rule + override action
toxicity_rule = Rule(metric="input_toxicity", operator=RuleOperator.gt, target_value=0.01)
action = OverrideAction(
    type="OVERRIDE",
    choices=[
        "This input was blocked due to toxicity.",
        "Content flagged: please rephrase your input.",
        "Input rejected: it violated the safety policy.",
    ],
)

# Create a ruleset object
ruleset = Ruleset(rules=[toxicity_rule], action=action)

# Create a central Protect Stage without rulesets
stage = create_protect_stage(
    project_name=project_name,
    name="Toxicity Stage",
    stage_type=StageType.central,
)
print(f"✅ Created empty stage: {stage.name} (id={stage.id}) in project '{project_name}'.")

# Update the stage with SDK Ruleset objects
update_protect_stage(
    project_name=project_name,
    stage_name="Toxicity Stage",
    prioritized_rulesets=[ruleset],
)

print("✅ Updated stage with SDK Ruleset objects.")
