# Load environment and create a fresh project
load_dotenv()
project_name = os.environ.get("GALILEO_PROJECT")
if not project_name:
    raise RuntimeError("Please set GALILEO_PROJECT in your environment or .env file.")

# Ensure the project name is new for this tutorial
if get_project(name=project_name):
    raise RuntimeError(
        f"Project '{project_name}' already exists.\n"
        "➡️ For this tutorial, update GALILEO_PROJECT in your .env file to a new, unused name.\n"
        "The script will then create the project for you automatically."
    )

# Create the project
project = create_project(name=project_name)
print(f"✅ Created project: {project.name} (id={project.id})")
