```python Python (Beta)
from galileo.__future__ import Dataset

# Get an existing dataset
dataset = Dataset.get(name="countries")

# Get the version history
versions = dataset.get_version_history()

# List out the rows added with each version
for version in versions:
    print(f"""
    Version index: {version.get('version_index')},
    rows added: {version.get('rows_added')}
    """)

# Get specific version of the dataset
specific_version = dataset.get_version(index=1)
```
