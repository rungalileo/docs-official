```python Python (Beta)
from galileo.__future__ import Dataset

# Get an existing dataset
dataset = Dataset.get(name="countries")

# Add new rows to the dataset
dataset.add_rows([
    {
        "input": "Which continent is Morocco in?",
        "output": "Africa",
    },
    {
        "input": "Which continent is Australia in?",
        "output": "Oceania",
    },
])
```
